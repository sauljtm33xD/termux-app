COMANDOS RÁPIDOS SIN MENÚ
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Instalar lo básico (curl, git, python, nodejs):
    termux-manager install-basic

Ver información del sistema:
    termux-manager system-info

Ver herramientas de la comunidad:
    termux-manager install-community

Información sobre acceso root:
    termux-manager root-info

Optimizar Termux (limpiar y actualizar):
    termux-manager optimize

PROBLEMAS COMUNES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

❌ "curl: not found"
Solución: Instalar curl primero
    apt update && apt install curl

❌ "No hay conexión"
Solución: Verificar WiFi o datos, luego:
    ping 8.8.8.8

❌ "Permiso denegado"
Solución:
    chmod +x ~/.termux-manager/termux-manager
